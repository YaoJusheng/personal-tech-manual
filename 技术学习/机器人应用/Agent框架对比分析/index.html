    📋 概述¶

    AI Agent是由大语言模型驱动的智能体，具备自主决策、工具调用和协作能力，是2026年最热门的"软件机器人"开发方向。本文档对主流AI Agent框架进行深度对比分析，提供详细的示例代码和生产级解决方案。

    🎯 文档定位¶

    • 技术选型：为AI Agent项目选择合适的框架
    • 架构设计：理解各框架的核心能力和设计模式
    • 实施指导：提供最小示例和最佳实践
    • 生产部署：包含容器化、监控和运维方案

    🧠 一、核心框架深度对比¶

    🔥 LangGraph - 状态机驱动的复杂工作流¶

    核心特性¶

    LangGraph是LangChain生态系统中的高级框架，专门用于构建复杂的多步骤工作流，具有强大的状态管理和图编排能力。

    特性 描述 优势 劣势
    图结构编排 支持复杂的节点连接和条件分支 灵活的工作流设计 学习曲线陡峭
    状态持久化 自动保存和恢复工作流状态 支持长时间运行的任务 需要数据库支持
    人工介入 支持在关键节点暂停等待人工确认 提高任务准确性 增加复杂度
    错误处理 内置错误处理和重试机制 提高系统稳定性 配置复杂

    最佳适用场景¶

    • 金融审批流程：需要严格控制执行路径和人工审核
    • 医疗诊断系统：需要多步骤推理和专家确认
    • 法律文档处理：复杂的文档审查和审批流程
    • 多轮对话系统：需要状态管理的复杂对话

    完整示例代码¶

    # langgraph_financial_approval.py
from langgraph.graph import Graph, END, StateGraph
from langgraph.prebuilt import ToolExecutor
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, AIMessage, SystemMessage
from typing import TypedDict, List, Annotated
import operator

# 定义状态结构
class AgentState(TypedDict):
    messages: Annotated[List[HumanMessage | AIMessage], operator.add]
    current_step: str
    approved_by: List[str]
    total_amount: float
    risk_level: str

# 定义工具
@tool
def check_credit_score(customer_id: str) -> str:
    """检查客户信用评分"""
    # 模拟信用检查
    credit_scores = {"C001": "750", "C002": "620", "C003": "580"}
    return f"客户{customer_id}信用评分：{credit_scores.get(customer_id, '未知')}"

@tool
def calculate_risk(amount: float, credit_score: str) -> str:
    """计算风险等级"""
    score = int(credit_score)
    if amount < 10000 and score > 700:
        return "低风险"
    elif amount < 50000 and score > 600:
        return "中风险"
    else:
        return "高风险"

@tool
def get_approval_officer(risk_level: str) -> str:
    """获取审批负责人"""
    officers = {
        "低风险": "初级审批官",
        "中风险": "中级审批官",
        "高风险": "高级审批官"
    }
    return officers.get(risk_level, "系统自动审批")

# 创建LLM
llm = ChatOpenAI(model="gpt-4o", temperature=0)

# 定义节点函数
def initial_assessment(state: AgentState):
    """初步评估节点"""
    messages = state["messages"]
    last_message = messages[-1]

    # 解析客户信息
    content = last_message.content
    customer_id = "C001"  # 从内容中解析
    amount = 25000        # 从内容中解析

    # 检查信用评分
    credit_result = check_credit_score(customer_id)
    credit_score = credit_result.split("：")[1]

    # 计算风险等级
    risk_level = calculate_risk(amount, credit_score)

    # 更新状态
    state["current_step"] = "credit_check"
    state["total_amount"] = amount
    state["risk_level"] = risk_level

    response = f"""
    初步评估完成：
    - 客户ID: {customer_id}
    - 申请金额: ¥{amount:,.2f}
    - 信用评分: {credit_score}
    - 风险等级: {risk_level}
    """

    return {
        "messages": [AIMessage(content=response)],
        **state
    }

def human_approval(state: AgentState):
    """人工审批节点"""
    risk_level = state["risk_level"]
    officer = get_approval_officer(risk_level)

    state["current_step"] = "human_approval"

    response = f"""
    需要人工审批：
    - 风险等级: {risk_level}
    - 审批负责人: {officer}
    - 申请金额: ¥{state['total_amount']:,.2f}

    请输入审批结果（approve/reject）：
    """

    return {
        "messages": [AIMessage(content=response)],
        **state
    }

def final_decision(state: AgentState):
    """最终决策节点"""
    messages = state["messages"]
    last_message = messages[-1]

    if "approve" in last_message.content.lower():
        decision = "批准"
        state["approved_by"].append(get_approval_officer(state["risk_level"]))
    else:
        decision = "拒绝"

    state["current_step"] = "final_decision"

    response = f"""
    审批决策：
    - 结果: {decision}
    - 审批流程: {" → ".join(state["approved_by"])}
    - 最终状态: {decision}
    """

    return {
        "messages": [AIMessage(content=response)],
        **state
    }

# 创建图
workflow = StateGraph(AgentState)

# 添加节点
workflow.add_node("initial_assessment", initial_assessment)
workflow.add_node("human_approval", human_approval)
workflow.add_node("final_decision", final_decision)

# 添加边
workflow.add_edge("initial_assessment", "human_approval")
workflow.add_edge("human_approval", "final_decision")
workflow.add_edge("final_decision", END)

# 设置入口点
workflow.set_entry_point("initial_assessment")

# 编译图
app = workflow.compile()

# 使用示例
if __name__ == "__main__":
    initial_state = {
        "messages": [HumanMessage(content="客户C001申请贷款25000元")],
        "current_step": "start",
        "approved_by": [],
        "total_amount": 0,
        "risk_level": ""
    }

    result = app.invoke(initial_state)
    print("=== 审批流程完成 ===")
    for msg in result["messages"]:
        print(f"{msg.content}")

class CollaborativeTask(Task):
    """支持Agent间协作的任务"""

    def __init__(self, description, agent, dependencies=None, **kwargs):
        super().__init__(description, agent, **kwargs)
        self.dependencies = dependencies or []
        self.shared_context = {}

    def add_dependency(self, task_id):
        """添加任务依赖"""
        self.dependencies.append(task_id)

    def update_shared_context(self, key: str, value: Any):
        """更新共享上下文"""
        self.shared_context[key] = value

    def get_shared_context(self, key: str) -> Any:
        """获取共享上下文"""
        return self.shared_context.get(key)

    """监控Agent，负责收集和上报指标"""

    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.start_time = time.time()

    def record_request(self, agent_type: str, status: str, duration: float):
        """记录请求指标"""
        REQUEST_COUNT.labels(agent_type=agent_type, status=status).inc()
        REQUEST_DURATION.observe(duration)

    def record_session(self, session_id: str, action: str):
        """记录会话指标"""
        if action == "start":
            ACTIVE_SESSIONS.inc()
        elif action == "end":

        """记录Token使用量"""
        TOKEN_USAGE.labels(model_type=model_type, agent_type=agent_type).inc(tokens)

    def record_error(self, error_type: str, agent_type: str):
        """记录错误指标"""
        ERROR_COUNT.labels(error_type=error_type, agent_type=agent_type).inc()

    def get_system_metrics(self) -> Dict[str, Any]:
        """获取系统指标"""
        return {
            "uptime": time.time() - self.start_time,
            "active_sessions": ACTIVE_SESSIONS._value.get(),

    # error_handling.py
import time
import logging
from functools import wraps
from typing import Callable, Any, Optional
import random

class RetryHandler:
    """重试处理器"""

    def __init__(self, max_retries: int = 3, backoff_factor: float = 1.0):
        self.max_retries = max_retries
        self.backoff_factor = backoff_factor
        self.logger = logging.getLogger(__name__)

    def retry(self, func: Callable) -> Callable:
        """重试装饰器"""
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(self.max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt < self.max_retries - 1:
                        wait_time = self.backoff_factor * (2 ** attempt)
                        self.logger.warning(
                            f"第{attempt + 1}次尝试失败，{wait_time}秒后重试: {str(e)}"
                        )
                        time.sleep(wait_time)
                    else:
                        self.logger.error(f"重试次数已达上限，放弃执行: {str(e)}")
            raise last_exception
        return wrapper

import time
import statistics
from typing import List, Dict, Any

class PerformanceBenchmark:
    """性能基准测试"""

    def __init__(self):
        self.results = {}

    def benchmark_agent(self, agent_name: str, func, *args, **kwargs):
        """测试Agent性能"""
        times = []
        for _ in range(10):  # 运行10次取平均值
            start_time = time.time()
            result = func(*args, **kwargs)
            end_time = time.time()
            times.append(end_time - start_time)

        avg_time = statistics.mean(times)
        std_time = statistics.stdev(times)

        self.results[agent_name] = {
            "avg_time": avg_time,
            "std_time": std_time,
            "min_time": min(times),
            "max_time": max(times)
        }

        return result

    def get_results(self) -> Dict[str, Any]:
        """获取测试结果"""
        return self.results

    def print_results(self):
        """打印测试结果"""
        print("=== 性能基准测试结果 ===")
        for agent_name, metrics in self.results.items():
            print(f"{agent_name}:")
            print(f"  平均时间: {metrics['avg_time']:.2f}秒")
            print(f"  标准差: {metrics['std_time']:.2f}秒")
            print(f"  最小时间: {metrics['min_time']:.2f}秒")
            print(f"  最大时间: {metrics['max_time']:.2f}秒")
            print()
