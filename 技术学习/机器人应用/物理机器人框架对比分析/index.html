    一、概览¶

    1.1 文档定位¶

    本文档对当下主流的机器人应用开发框架进行系统性对比分析，帮助开发者根据项目需求选择合适的技术方案，并提供最小示例和使用场景推荐。

    1.2 适用场景¶

    • 新项目选型：为机器人项目选择合适的技术栈
    • 技术迁移评估：从ROS1或其他框架迁移到ROS2的可行性分析
    • 生产部署决策：根据业务需求选择合适的部署方案
    • 团队技能规划：制定团队技术学习路径

    1.3 评估维度¶

    评估维度 具体指标 重要性
    架构设计 模块化程度、扩展性、耦合度 ⭐⭐⭐⭐⭐

import rclpy
from rclpy.node import Node
from std_msgs.msg import String

class Talker(Node):
    def __init__(self):
        super().__init__('talker')
        self.publisher = self.create_publisher(String, 'chatter', 10)
        self.timer = self.create_timer(1.0, self.timer_callback)
        self.count = 0

        msg = String()
        msg.data = f'Hello ROS2: {self.count}'
        self.publisher.publish(msg)
        self.get_logger().info(f'Publishing: {msg.data}')
        self.count += 1

def main():
    rclpy.init()
    talker = Talker()
    rclpy.spin(talker)
    talker.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()

#include "rclcpp/rclcpp.hpp"
#include "std_msgs/msg/string.hpp"

class Talker : public rclcpp::Node {
public:
    Talker() : Node("talker"), count_(0) {
        publisher_ = this->create_publisher<std_msgs::msg::String>("chatter", 10);
        timer_ = this->create_wall_timer(
            std::chrono::seconds(1),
            std::bind(&Talker::timer_callback, this));
    }

private:
    void timer_callback() {
        auto message = std_msgs::msg::String();
        message.data = "Hello ROS2: " + std::to_string(count_++);
        RCLCPP_INFO(this->get_logger(), "Publishing: '%s'", message.data.c_str());
        publisher_->publish(message);
    }
    rclcpp::TimerBase::SharedPtr timer_;
    rclcpp::Publisher<std_msgs::msg::String>::SharedPtr publisher_;
    size_t count_;
};

int main(int argc, char* argv[]) {
    rclcpp::init(argc, argv);
    auto node = std::make_shared<Talker>();
    rclcpp::spin(node);
    rclcpp::shutdown();
    return 0;
}

    4.2.3 客户端实现¶

    # client.py
import sys
import rclpy
from rclpy.node import Node
from example_interfaces.srv import AddTwoInts

class MinimalClient(Node):

    def __init__(self):
        super().__init__('minimal_client')
        self.cli = self.create_client(AddTwoInts, 'add_two_ints')
        while not self.cli.wait_for_service(timeout_sec=1.0):
            self.get_logger().info('service not available, waiting again...')
        self.req = AddTwoInts.Request()

    def send_request(self, a, b):
        self.req.a = a
        self.req.b = b
        self.future = self.cli.call_async(self.req)
        rclpy.spin_until_future_complete(self, self.future)
        return self.future.result()

def main(args=None):
    rclpy.init()
    minimal_client = MinimalClient()
    if len(sys.argv) != 3:
        minimal_client.get_logger().info('Usage: python client.py <number> <number>')
        sys.exit(1)

    a = int(sys.argv[1])
    b = int(sys.argv[2])
    minimal_client.get_logger().info('Requesting sum of %d + %d' % (a, b))
    response = minimal_client.send_request(a, b)
    minimal_client.get_logger().info('Result of add_two_ints: %d' % response.sum)
    minimal_client.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()

    # fibonacci_action_server.py
import rclpy
from rclpy.action import ActionServer
from rclpy.node import Node
from action_tutorials_interfaces.action import Fibonacci

class FibonacciActionServer(Node):

    def __init__(self):
        super().__init__('fibonacci_action_server')
        self._action_server = ActionServer(
            self,
            Fibonacci,
            'fibonacci',
            self.execute_callback)

    def execute_callback(self, goal_handle):
        self.get_logger().info('Executing goal...')

        feedback_msg = Fibonacci.Feedback()
        feedback_msg.partial_sequence = [0, 1]

        for i in range(1, goal_handle.request.order):
            feedback_msg.partial_sequence.append(
                feedback_msg.partial_sequence[i] + feedback_msg.partial_sequence[i-1])

            goal_handle.publish_feedback(feedback_msg)
            self.get_logger().info('Publishing feedback: {0}'.format(feedback_msg.partial_sequence))

        goal_handle.succeed()
        result = Fibonacci.Result()
        result.sequence = feedback_msg.partial_sequence
        self.get_logger().info('Returning result: {0}'.format(result.sequence))
        return result

def main(args=None):
    rclpy.init()
    fibonacci_action_server = FibonacciActionServer()
    rclpy.spin(fibonacci_action_server)
    rclpy.shutdown()

if __name__ == '__main__':
    main()

    九点五、ROS2 + AI Agent 融合（2026 核心趋势）¶

    2026 年物理机器人最重要的趋势是 "LLM 当大脑、ROS2 当手脚"：用大语言模型做高层语义理解与任务规划，用 ROS2 做实时硬件控制与底层通信。

    9.5.1 分层架构¶

    flowchart TB
    subgraph Brain["大脑层（LLM / AI Agent）"]
        A1[自然语言指令] --> A2[LLM 理解意图]
        A2 --> A3[任务规划/分解]
        A3 --> A4[选择技能/工具]
    end
    subgraph Bridge["桥接层（MCP / 工具调用）"]
        B1[ROS2 动作封装为工具]
        B2[感知结果回传 LLM]
    end
    subgraph Body["手脚层（ROS2 实时控制）"]
        C1[Nav2 导航]
        C2[MoveIt2 机械臂]
        C3[感知/SLAM]
        C4[ros2_control 底盘]
    end
    Brain --> Bridge
    Bridge --> Body
    Body -.传感器反馈.-> Bridge
    Bridge -.观测结果.-> Brain
    层次 职责 技术 实时性要求
    大脑层 语义理解、任务规划、决策 LangGraph / CrewAI + LLM 低（秒级）
    桥接层 工具封装、结果回传、协议转换 MCP / 自定义 Bridge 中
    手脚层 导航、运动、感知、控制 ROS2 + Nav2/MoveIt2 高（毫秒级）

    关键设计原则：实时控制环绝不交给 LLM。LLM 只下发高层目标（"去厨房拿杯子"），ROS2 负责把目标翻译成毫秒级的电机指令。LLM 的不确定性与延迟必须被隔离在控制环之外。

    9.5.2 典型应用场景¶

    场景 LLM 负责 ROS2 负责
    语音指令导航 理解"去会议室"，规划路径点 Nav2 执行避障导航
    自然语言抓取 识别"拿起红色杯子"，定位目标 MoveIt2 规划抓取轨迹
    异常自主决策 "检测到障碍，绕行还是等待？" 感知上报，执行绕行
    多机协同调度 分配任务给不同机器人 各机器人 ROS2 栈执行

    9.5.3 生产级集成示例（LangGraph 大脑 + ROS2 手脚）¶

    将 ROS2 的导航能力封装为 LLM 可调用的工具，由 LangGraph 编排决策：

    # llm_ros2_bridge.py
# 大脑：LangGraph 规划；手脚：ROS2 Nav2 执行
import rclpy
from rclpy.node import Node
from rclpy.action import ActionClient
from nav2_msgs.action import NavigateToPose
from geometry_msgs.msg import PoseStamped
from langchain_core.tools import tool
from langchain_openai import ChatOpenAI


class Nav2Bridge(Node):
    """把 Nav2 导航封装成可被 LLM 调用的桥接节点"""

    def __init__(self):
        super().__init__('nav2_bridge')
        self._client = ActionClient(self, NavigateToPose, 'navigate_to_pose')
        # 预定义语义地点 → 坐标（生产中来自地图标注/数据库）
        self.locations = {
            "会议室": (5.0, 3.0),
            "厨房": (8.0, 1.0),
            "充电桩": (0.0, 0.0),
        }

    def navigate_to(self, location_name: str) -> str:
        """执行导航到指定语义地点（实时控制由 Nav2 完成）"""
        if location_name not in self.locations:
            return f"未知地点: {location_name}"

        x, y = self.locations[location_name]
        goal = NavigateToPose.Goal()
        goal.pose.header.frame_id = 'map'
        goal.pose.pose.position.x = x
        goal.pose.pose.position.y = y
        goal.pose.pose.orientation.w = 1.0

        self._client.wait_for_server(timeout_sec=5.0)
        future = self._client.send_goal_async(goal)
        rclpy.spin_until_future_complete(self, future)
        goal_handle = future.result()
        if not goal_handle.accepted:
            return f"导航目标被拒绝: {location_name}"

        result_future = goal_handle.get_result_async()
        rclpy.spin_until_future_complete(self, result_future)
        return f"已到达{location_name}"


# 全局桥接实例
_bridge: Nav2Bridge | None = None


@tool
def go_to_location(location: str) -> str:
    """让机器人移动到指定地点。location 可选：会议室、厨房、充电桩"""
    return _bridge.navigate_to(location)


def main():
    global _bridge
    rclpy.init()
    _bridge = Nav2Bridge()

    # LLM 大脑：理解自然语言并决定调用导航工具
    llm = ChatOpenAI(model="gpt-4o").bind_tools([go_to_location])
    user_cmd = "我开完会了，先去厨房，然后回去充电"

    response = llm.invoke(user_cmd)
    # LLM 输出工具调用序列，逐个执行（控制环在 ROS2 侧）
    for call in response.tool_calls:
        result = go_to_location.invoke(call["args"])
        _bridge.get_logger().info(f"执行结果: {result}")

    _bridge.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()

    生产落地要点：

    • MCP 标准化：将 ROS2 能力暴露为 MCP Server，使 LLM 工具调用跨框架复用（详见 LLM 智能体调度模型）。
    • 安全护栏：LLM 下发的目标必须经过规则校验（如禁区检查、速度上限）后才交给 ROS2 执行。
    • 降级策略：LLM/网络不可用时，机器人退回到纯 ROS2 的预设行为（如返回充电桩），保证基本安全。
    • 延迟隔离：LLM 决策走异步，绝不阻塞 ROS2 的实时控制回调。
